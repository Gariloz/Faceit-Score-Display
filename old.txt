    function refreshSelectors() {
        // Перезапускаем MutationObserver с новыми элементами
        if (isScriptActive) {
            // Находим новый контейнер для наблюдения
            let newObserveTarget = null;
            for (const containerSelector of SELECTORS.SCORE_CONTAINERS) {
                newObserveTarget = document.querySelector(containerSelector);
                if (newObserveTarget) break;
            }
            newObserveTarget = newObserveTarget || document.body;
            
            // Всегда пересоздаем observer для максимальной надежности
            if (currentObserver) {
                currentObserver.disconnect();
            }
            currentObserveTarget = newObserveTarget;
            currentObserver = new MutationObserver(handleDOMChanges);
            currentObserver.observe(currentObserveTarget, { 
                childList: true, 
                subtree: true, 
                attributes: true 
            });
            
            // Принудительно обновляем счет после обновления селекторов
            setTimeout(() => {
                if (isScriptActive) {
                    updateScore();
                }
            }, 100);
        }
    }

    // === ПОИСК ЭЛЕМЕНТОВ СЧЕТА И ВРЕМЕНИ ===
    function findScores() {
        // Проверяем что мы на странице матча, а не на главной
        const currentUrl = window.location.href;
        const isMatchPage = CONFIG.MATCH_URL_PATTERNS.some(pattern => currentUrl.includes(pattern));
        
        if (!isMatchPage) {
            // Если не на странице матча, возвращаем пустые элементы
            return {
                scoreElements: { scoreTeam1Element: null, scoreTeam2Element: null },
                timeElement: null
            };
        }
        
        let scoreElements = null;
        let timeElement = null;

        // 1. Основной поиск по всем селекторам счета
        for (const selector of SELECTORS.SCORE_ELEMENTS) {
            const scoreNodes = document.querySelectorAll(selector);
        if (scoreNodes.length >= 2) {
                scoreElements = { 
                scoreTeam1Element: scoreNodes[0], 
                scoreTeam2Element: scoreNodes[1] 
            };
                break;
            }
        }

        // 2. Поиск внутри контейнеров для счета
        if (!scoreElements) {
            for (const containerSelector of SELECTORS.SCORE_CONTAINERS) {
                const container = document.querySelector(containerSelector);
        if (container) {
                    for (const selector of SELECTORS.SCORE_ELEMENTS) {
                        const scoreNodes = container.querySelectorAll(selector);
            if (scoreNodes.length >= 2) {
                            scoreElements = { 
                    scoreTeam1Element: scoreNodes[0], 
                    scoreTeam2Element: scoreNodes[1] 
                };
                            break;
                        }
                    }
                    if (scoreElements) break;
                }
            }
        }

        // 3. Резервный поиск по содержимому и классам для счета
        if (!scoreElements) {
            const fallbackNodes = Array.from(document.querySelectorAll('h3, div, span'))
                .filter(el => {
                    const text = el.textContent?.trim() || '';
                    const hasNumber = /\b\d+\b/.test(text);
                    const hasRelevantClass = SELECTORS.FALLBACK_CLASS_PATTERNS.some(pattern => 
                        el.className.includes(pattern)
                    );
                    return hasNumber && hasRelevantClass;
                });

            if (fallbackNodes.length >= 2) {
                scoreElements = { 
                    scoreTeam1Element: fallbackNodes[0], 
                    scoreTeam2Element: fallbackNodes[1] 
                };
            }
        }

        // 4. Поиск времени матча
        for (const selector of SELECTORS.TIME_ELEMENTS) {
            const timeNode = document.querySelector(selector);
            if (timeNode && timeNode.textContent.trim()) {
                timeElement = timeNode;
                break;
            }
        }

        // 4.1. Специальный поиск в контейнере FactionsDetails
        if (!timeElement) {
            const factionsContainer = document.querySelector('[class*="FactionsDetails__Details"]');
            if (factionsContainer) {
                const timeInContainer = factionsContainer.querySelector('.Tooltip__TriggerContainer');
                if (timeInContainer && timeInContainer.textContent.trim()) {
                    timeElement = timeInContainer;
                }
            }
        }

        // 5. Резервный поиск времени по содержимому (формат MM:SS или HH:MM:SS)
        if (!timeElement) {
            const timeNodes = Array.from(document.querySelectorAll('*'))
            .filter(el => {
                const text = el.textContent?.trim() || '';
                // Поддерживаем форматы: MM:SS, HH:MM:SS, и точный формат FACEIT 00:00:53
                const timePattern = /^\d{2}:\d{2}:\d{2}$|^\d{1,2}:\d{2}$/;
                return timePattern.test(text) && el.children.length === 0;
            });
            
            if (timeNodes.length > 0) {
                timeElement = timeNodes[0];
            }
        }

        // Если ничего не найдено, попробуем принудительно обновить селекторы
        if (!scoreElements && !timeElement && isScriptActive) {
            refreshSelectors();
        }
        
        // Дополнительная проверка - если счет не изменился долго, обновляем селекторы
        if (scoreElements && isScriptActive) {
            const currentScore = `${scoreElements.scoreTeam1Element.textContent.trim()}-${scoreElements.scoreTeam2Element.textContent.trim()}`;
            if (currentScore === lastScore) {
            // Если счет не меняется больше заданного времени, обновляем селекторы
            if (!window.lastScoreUpdate) window.lastScoreUpdate = Date.now();
            if (Date.now() - window.lastScoreUpdate > CONFIG.SCORE_STUCK_CHECK_INTERVAL) {
                    refreshSelectors();
                    window.lastScoreUpdate = Date.now();
                }
            } else {
                window.lastScoreUpdate = Date.now();
            }
        }

        return {
            scoreElements: scoreElements || { scoreTeam1Element: null, scoreTeam2Element: null },
            timeElement
        };
    }

    // === ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ ===
    function isPopupAlive() {
        try {
            return !!(scoreWindow && !scoreWindow.closed);
        } catch {
            return false;
        }
    }

    function getSettings() {
        return {
            fontSize: Number(localStorage.getItem(STORAGE_KEYS.FONT_SIZE) || CONFIG.FONT_SIZE),
            soundEnabled: localStorage.getItem(STORAGE_KEYS.SOUND) === '1'
        };
    }

    // === ОБНОВЛЕНИЕ СЧЕТА ===
    function sendScoreToPopup(fontSize, playSound) {
        const { scoreElements, timeElement } = findScores();
        
        // Если элементы не найдены, отправляем "Ожидание..."
        if (!scoreElements || !scoreElements.scoreTeam1Element || !scoreElements.scoreTeam2Element) {
            if (isPopupAlive()) {
                updatePopupDisplay('Ожидание...', 'Ожидание...', fontSize, 'Ожидание...');
            }
            broadcastScoreUpdate('Ожидание...', 'Ожидание...', fontSize, 'Ожидание...');
            return;
        }

        const scoreTeam1 = scoreElements.scoreTeam1Element.textContent.trim() || '0';
        const scoreTeam2 = scoreElements.scoreTeam2Element.textContent.trim() || '0';
        const currentScore = `${scoreTeam1}-${scoreTeam2}`;
        const matchTime = timeElement ? timeElement.textContent.trim() : '';

        // Проверяем изменение счета для звукового уведомления
        const scoreChanged = currentScore !== lastScore;
        if (scoreChanged && playSound && currentScore !== lastSoundScore) {
            lastScore = currentScore;
            lastSoundScore = currentScore; // Запоминаем счет для которого воспроизвели звук
            
            // Разблокируем аудио при каждом изменении счета для гарантированной работы звука
            if (isPopupAlive()) {
                try {
                    scoreWindow.postMessage({ type: 'unlockAudio' }, '*');
                } catch {}
            }
            
            // Отправляем сигнал о воспроизведении звука в попап
            if (isPopupAlive()) {
                try {
                    scoreWindow.postMessage({ type: 'playSound', score: currentScore }, '*');
                } catch {}
            }
            
            // Дополнительно отправляем сигнал звука через BroadcastChannel
            if (scoreChannel) {
                try {
                    scoreChannel.postMessage({ type: 'playSound', score: currentScore });
                } catch {}
            }
        } else if (scoreChanged) {
            lastScore = currentScore;
            // Сбрасываем lastSoundScore при изменении счета, чтобы звук мог воспроизводиться снова
            lastSoundScore = null;
        }

        // Обновляем попап если он открыт
        if (isPopupAlive()) {
            updatePopupDisplay(scoreTeam1, scoreTeam2, fontSize, matchTime);
        }

        // Отправляем данные через каналы связи
        broadcastScoreUpdate(scoreTeam1, scoreTeam2, fontSize, matchTime);
    }

    function updatePopupDisplay(scoreTeam1, scoreTeam2, fontSize, matchTime = '') {
        try {
            const doc = scoreWindow.document;
            const scoreDisplay = doc.getElementById('scoreDisplay');
            const timeDisplay = doc.getElementById('timeDisplay');
            const fontSizeInput = doc.getElementById('fontSizeInput');
            
            if (scoreDisplay) {
                scoreDisplay.textContent = `${scoreTeam1} - ${scoreTeam2}`;
                scoreDisplay.style.fontSize = `${fontSize}px`;
            }
            if (timeDisplay && matchTime) {
                timeDisplay.textContent = matchTime;
                timeDisplay.style.display = 'block';
            } else if (timeDisplay) {
                timeDisplay.style.display = 'none';
            }
            if (fontSizeInput) {
                fontSizeInput.value = String(fontSize);
            }
        } catch {}
    }

    function broadcastScoreUpdate(scoreTeam1, scoreTeam2, fontSize, matchTime = '') {
        try {
            scoreChannel?.postMessage({ type: 'update', scoreTeam1, scoreTeam2, fontSize, matchTime });
            localStorage.setItem(STORAGE_KEYS.SCORE_PAYLOAD, 
                JSON.stringify({ t: Date.now(), scoreTeam1, scoreTeam2, fontSize, matchTime }));
        } catch {}
    }

    // === УПРАВЛЕНИЕ ПОПАПОМ ===
    function ensurePopupContent() {
        if (!isPopupAlive()) return false;
        
        try {
            const hasContent = !!scoreWindow.document.getElementById('scoreDisplay');
            if (!hasContent) {
                return writePopupHTML(scoreWindow);
            }
            return true;
        } catch {
            return false;
        }
    }

    function startPopupKeepAlive() {
        clearInterval(popupKeepAliveTimer);
        popupKeepAliveTimer = setInterval(() => {
            if (!isPopupAlive()) return;
            
            try {
                const hasContent = !!scoreWindow.document.getElementById('scoreDisplay');
                if (!hasContent) {
                    writePopupHTML(scoreWindow);
                }
            } catch {}
        }, 1000);
    }

    // === АВТООБНОВЛЕНИЕ СТРАНИЦЫ ===
    function applyAutoReloadPolicy() {
        clearInterval(autoReloadTimer);
        try {
            const enabled = localStorage.getItem(STORAGE_KEYS.AUTO_RELOAD_ENABLED) === '1';
            const seconds = Number(localStorage.getItem(STORAGE_KEYS.AUTO_RELOAD_SECONDS) || '0');
            const ms = seconds * 1000;
            
            if (enabled && ms >= 5000) {
                autoReloadTimer = setInterval(() => location.reload(), ms);
            }
        } catch {}
    }


    // === ОБНОВЛЕНИЕ СЧЕТА ===
    function updateScore() {
        // Работаем только если скрипт активирован
        if (!isScriptActive) return;
        
        if (isUpdating) return;
        isUpdating = true;

        const settings = getSettings();
        sendScoreToPopup(settings.fontSize, settings.soundEnabled);
        
        // Дополнительная проверка - если счет не обновляется, принудительно обновляем селекторы
        setTimeout(() => {
            if (isScriptActive) {
                const { scoreElements } = findScores();
                if (!scoreElements || !scoreElements.scoreTeam1Element || !scoreElements.scoreTeam2Element) {
                    refreshSelectors();
                }
            }
        }, CONFIG.UPDATE_CHECK_DELAY);

        isUpdating = false;
    }

    function startUpdateInterval() {
        clearInterval(updateInterval);
        clearInterval(fastCheckInterval);
        
        // Оптимизированный интервал обновления
        updateInterval = setInterval(updateScore, CONFIG.UPDATE_INTERVAL);
    }

    function stopUpdateInterval() {
        clearInterval(updateInterval);
        clearInterval(fastCheckInterval);
        clearInterval(selectorRefreshTimer);
        
        // Очищаем состояние скрипта при остановке
        localStorage.removeItem(STORAGE_KEYS.SCRIPT_ACTIVE);
        isScriptActive = false;
    }

    // === ПОСТОЯННОЕ ОБНОВЛЕНИЕ СЕЛЕКТОРОВ ДЛЯ АКТУАЛЬНОСТИ ===
    function continuousSelectorUpdate() {
        // Обновляем селекторы каждые 1 секунду для максимальной актуальности
        setInterval(() => {
            if (isScriptActive) {
                refreshSelectors();
            }
        }, 1000);
        
        // Дополнительно обновляем при каждом изменении DOM
        if (currentObserver) {
            currentObserver.disconnect();
        }
        
        // Пересоздаем observer с более агрессивными настройками
        currentObserveTarget = document.body; // Всегда наблюдаем весь body
        currentObserver = new MutationObserver((mutations) => {
            handleDOMChanges(mutations);
            // При любом изменении DOM принудительно обновляем селекторы
            setTimeout(() => {
                if (isScriptActive) {
                    refreshSelectors();
                }
            }, 100);
        });
        currentObserver.observe(currentObserveTarget, { 
            childList: true, 
            subtree: true, 
            attributes: true,
            characterData: true,
            attributeOldValue: true,
            characterDataOldValue: true
        });
    }

    function handleButtonClick() {
        // Активируем скрипт только при нажатии кнопки

        selectorRefreshTimer = setInterval(refreshSelectors, CONFIG.SELECTOR_REFRESH_INTERVAL);
        
        // Запускаем постоянное обновление селекторов для актуальности
        continuousSelectorUpdate();
    }
